    class ScMetaModel(SQLModel.__class__):
    """Custom class factory to auto-update table models.

    - Reuse description of the table and its fields as SQL comment.

        Checking if the table and its fields have explicit comment set
        to be shown in the `CREATE TABLE` statements, and if not,
        reuse the optional table and field descriptions. Table
        docstrings are truncated to first line.

    - Reuse description of the fields to dynamically append to the
        docstring in the Attributes section.

    - Set `__validator__` to the parent Pydantic model without
        `table=True`, which is useful for running validations.
        The Pydantic model is found by the parent class' name ending in "Base".

    - Auto-generate SCD table docs from the non-SCD table docs.
    """

    def __init__(subclass, *args, **kwargs):
        super().__init__(*args, **kwargs)
        # early return for non-tables
        if subclass.model_config.get("table") is None:
            return
        satable = subclass.metadata.tables[subclass.__tablename__]

        # enforce auto-naming constrains as per
        # https://alembic.sqlalchemy.org/en/latest/naming.html
        subclass.metadata.naming_convention = {
            "ix": "ix_%(column_0_label)s",
            "uq": "uq_%(table_name)s_%(column_0_name)s",
            "ck": "ck_%(table_name)s_%(constraint_name)s",
            "fk": "fk_%(table_name)s_%(column_0_name)s_%(referred_table_name)s",
            "pk": "pk_%(table_name)s",
        }

        # table comment
        if subclass.__doc__ and satable.comment is None:
            satable.comment = subclass.__doc__.splitlines()[0]

        # column comments
        for k, v in subclass.model_fields.items():
            comment = satable.columns[k].comment
            if v.description and comment is None:
                satable.columns[k].comment = v.description

        # generate docstring for SCD tables
        if subclass.__name__.endswith("Scd"):
            from .tables import tables

            nonscd = [t for t in tables if t.__name__ == subclass.__name__[:-3]][0]
            doclines = nonscd.__doc__.splitlines()
            # drop trailing dot and append SCD
            doclines[0] = doclines[0][:-1] + " (SCD Type 2)."
            subclass.__doc__ = "\n".join(doclines)
        else:
            # describe table columns as attributes in docstring
            subclass.__doc__ = subclass.__doc__ + "\n\nAttributes:\n"
            for k, v in subclass.model_fields.items():
                if not hasattr(v.annotation, "__args__"):
                    typehint = v.annotation.__name__
                else:
                    typehint = str(v.annotation)
                description = satable.columns[k].comment
                subclass.__doc__ = (
                    subclass.__doc__ + f"    {k} ({typehint}): {description}\n"
                )

        # find Pydantic model parent to be used for validating
        subclass.__validator__ = [
            m for m in subclass.__bases__ if m.__name__.endswith("Base")
        ][0]

    class ScModel(SQLModel, metaclass=ScMetaModel):
    """Custom extensions to SQLModel objects and tables.

    Extra features:

    - auto-generated table names using [snake_case][sc_crawler.str_utils.snake_case],
    - support for hashing table rows,
    - reuse description field of tables/columns as SQL comment,
    - reuse description field of columns to extend the `Attributes` section of the docstring.
    """

    @declared_attr  # type: ignore
    def __tablename__(cls) -> str:
        """Override tables names using all-lowercase [snake_case][sc_crawler.str_utils.snake_case]."""
        return snake_case(cls.__name__)

    @classmethod
    def get_columns(cls) -> List[str]:
        """Return the table's column names in a dict for all, primary keys, and attributes."""
        columns = cls.__table__.columns.keys()
        pks = [pk.name for pk in inspect(cls).primary_key]
        attributes = [a for a in columns if a not in set(pks)]
        return {"all": columns, "primary_keys": pks, "attributes": attributes}

    @classmethod
    def get_table_name(cls) -> str:
        """Return the SQLModel object's table name."""
        return str(cls.__tablename__)

    @classmethod
    def get_validator(cls) -> Union["ScModel", None]:
        """Return the parent Base Pydantic model (without a table definition)."""
        if cls.model_config.get("table") is None:
            return None
        return cls.__validator__

    @classmethod
    def get_scd(cls) -> Union["ScModel", None]:
        """Return the SCD version of the SQLModel table."""
        if cls.model_config.get("table") is None:
            return None
        from .tables_scd import tables_scd

        validator = cls.get_validator()
        scds = [t for t in tables_scd if t.get_validator() == validator]
        if len(scds) != 1:
            raise ValueError("Not found SCD definition.")
        return scds[0]

    @classmethod
    @cache
    def get_parent_tables(cls) -> Set[str]:
        """Names of the tables that the records of this table belong to.

        These are the tables referenced by a foreign key made up of primary
        key columns of this table, and (recursively) the parents of those
        tables. So a record cannot exist without its parent records, e.g. a
        server price without its vendor, region, zone and server.

        Examples:
            >>> from sc_crawler.tables import Zone
            >>> sorted(Zone.get_parent_tables())
            ['region', 'vendor']
        """
        from .utils import table_name_to_model

        parents = {
            constraint.referred_table.name
            for constraint in cls.__table__.foreign_key_constraints
            if all(column.primary_key for column in constraint.columns)
        }
        for parent in list(parents):
            parents |= table_name_to_model(parent).get_parent_tables()
        return parents

    @classmethod
    def get_key_filters(cls, rule_table_name: str, keys: dict) -> Optional[dict]:
        """Translate the `match` or `successor` of an archive rule into column filters on the table.

        This defines which records an [ArchiveRule][sc_crawler.tables.ArchiveRule]
        refers to, for both the SQL conditions and matching primary keys in
        Python. Tables referencing a parent without a foreign key override it.

        Args:
            rule_table_name: The `table_name` of the rule.
            keys: The `match` or `successor` of the rule.

        Returns:
            Dict of column names and values of the table, or `None` if the
                rule does not apply to the table.
        """
        if (
            cls.get_table_name() != rule_table_name
            and rule_table_name not in cls.get_parent_tables()
        ):
            return None
        return dict(keys)

    @classmethod
    def get_filters_condition(cls, filters: dict) -> ColumnElement:
        """SQL condition selecting the rows of the table with the given column values."""
        return and_(
            *[getattr(cls, column) == value for column, value in filters.items()]
        )

    @classmethod
    def matches_rule(cls, rule: "ArchiveRule", primary_key: dict) -> bool:
        """Check if an archive rule archives the record with the given primary key.

        Args:
            rule: An [ArchiveRule][sc_crawler.tables.ArchiveRule].
            primary_key: Primary key columns of the record with their values.
        """
        filters = cls.get_key_filters(rule.table_name, rule.match)
        if filters is None:
            return False
        return all(
            column in primary_key and primary_key[column] == value
            for column, value in filters.items()
        )

    @classmethod
    def is_archived(cls, primary_key: dict, rules: Optional[list] = None) -> bool:
        """Check if any of the archive rules archives the record with the given primary key.

        Args:
            primary_key: Primary key columns of the record with their values.
            rules: [ArchiveRule][sc_crawler.tables.ArchiveRule]s to check. Defaults to all current rules.
        """
        if rules is None:
            from .archive_rules import archive_rules as rules
        return any(cls.matches_rule(rule, primary_key) for rule in rules)

    @classmethod
    def archive_condition(cls, rules: Optional[list] = None) -> Optional[ColumnElement]:
        """SQL condition selecting the rows of the table archived by the rules.

        Child records are selected by the key columns of the rules instead of
        looking up their parents, so it also finds the child records of parents
        that are not in the database.

        Args:
            rules: [ArchiveRule][sc_crawler.tables.ArchiveRule]s to apply. Defaults to all current rules.

        Returns:
            SQLAlchemy condition, or `None` if no rule applies to the table.
        """
        if rules is None:
            from .archive_rules import archive_rules as rules
        conditions = []
        for rule in rules:
            filters = cls.get_key_filters(rule.table_name, rule.match)
            if filters is not None:
                conditions.append(cls.get_filters_condition(filters))
        return or_(*conditions) if conditions else None

    @classmethod
    def archived_by(cls, session: Session, primary_key: dict) -> list:
        """List the archive rules archiving a record, including the rules on its parent tables.

        Reads the rules from the `archive_rule` table of the database, so it also
        returns the rules that were removed from the code since (stored with
        `INACTIVE` status), and uses the same matching as `pull` and `sync`.
        It does not look up the record itself.

        Args:
            session: Database connection.
            primary_key: All primary key columns of the record with their values.
                Other keys (e.g. `observed_at` of an SCD record) are ignored.

        Returns:
            The matching [ArchiveRule][sc_crawler.tables.ArchiveRule]s, ordered
                by table name, `match` and `successor`.
        """
        from .tables import ArchiveRule

        columns = cls.get_columns()["primary_keys"]
        missing = [column for column in columns if column not in primary_key]
        if missing:
            raise ValueError(
                f"Missing primary key column(s) of {cls.get_table_name()}: "
                f"{', '.join(missing)}"
            )
        key = {}
        for column in columns:
            value = primary_key[column]
            # accept the name or value of enum members, which are stored by name
            column_type = cls.__table__.columns[column].type
            if isinstance(column_type, SaEnum) and column_type.enum_class is not None:
                enum_class = column_type.enum_class
                if isinstance(value, str) and value in enum_class.__members__:
                    value = enum_class[value]
                else:
                    value = enum_class(value)
            key[column] = value
        rules = [
            rule
            for rule in session.exec(select(ArchiveRule)).all()
            if cls.matches_rule(rule, key)
        ]
        return sorted(
            rules,
            key=lambda rule: (rule.table_name, str(rule.match), str(rule.successor)),
        )

    @classmethod
    def pk_key(cls, values: dict) -> str:
        """JSON dump of the primary key columns of a row, as used in [hash][sc_crawler.table_bases.ScModel.hash]."""
        pks = sorted(cls.get_columns()["primary_keys"])
        return dumps({pk: values.get(pk) for pk in pks}, sort_keys=True)

    @classmethod
    def hash(
        cls,
        session: Session,
        ignored: List[str] = ["observed_at"],
        progress: Optional[Progress] = None,
    ) -> dict:
        """Hash the content of the rows.

        Args:
            session: Database connection to use for object lookups.
            ignored: List of column names to exclude from hashing.
            progress: Optional progress bar to track the status of the hashing.

        Returns:
            Dictionary of the row hashes keyed by the JSON dump of primary keys.
        """
        pks = sorted(cls.get_columns()["primary_keys"])
        rows = session.exec(statement=select(cls))
        row_count = session.exec(select(func.count()).select_from(cls)).one()
        if progress:
            table_task_id = progress.add_task(
                cls.get_table_name(),
                total=row_count,
            )
        # no use of a generator as will need to serialize to JSON anyway
        hashes = {}
        for i, row in enumerate(rows):
            # NOTE Pydantic is warning when read Accelerator/Storage as dict
            # https://github.com/tiangolo/sqlmodel/issues/63#issuecomment-1081555082
            rowdict = row.model_dump(warnings=False)
            keys_id = cls.pk_key(rowdict)
            for dropkey in [*ignored, *pks]:
                rowdict.pop(dropkey, None)
            rowhash = sha1(dumps(rowdict, sort_keys=True).encode()).hexdigest()
            hashes[keys_id] = rowhash
            if progress:
                # updating the progress bar is expensive, so limit with manu iterations
                if row_count > 1e3:
                    if (i + 1) % 1000 == 0:
                        progress.update(table_task_id, advance=1000)
                    if i == row_count - 1:
                        progress.update(table_task_id, advance=row_count % 1000)
                else:
                    progress.update(table_task_id, advance=1)

        return hashes

    @classmethod
def archived_by(cls, session: Session, primary_key: dict) -> list:
    """List the archive rules archiving a record, including the rules on its parent tables.

    Reads the rules from the `archive_rule` table of the database, so it also
    returns the rules that were removed from the code since (stored with
    `INACTIVE` status), and uses the same matching as `pull` and `sync`.
    It does not look up the record itself.

    Args:
        session: Database connection.
        primary_key: All primary key columns of the record with their values.
            Other keys (e.g. `observed_at` of an SCD record) are ignored.

    Returns:
        The matching [ArchiveRule][sc_crawler.tables.ArchiveRule]s, ordered
            by table name, `match` and `successor`.
    """
    from .tables import ArchiveRule

    columns = cls.get_columns()["primary_keys"]
    missing = [column for column in columns if column not in primary_key]
    if missing:
        raise ValueError(
            f"Missing primary key column(s) of {cls.get_table_name()}: "
            f"{', '.join(missing)}"
        )
    key = {}
    for column in columns:
        value = primary_key[column]
        # accept the name or value of enum members, which are stored by name
        column_type = cls.__table__.columns[column].type
        if isinstance(column_type, SaEnum) and column_type.enum_class is not None:
            enum_class = column_type.enum_class
            if isinstance(value, str) and value in enum_class.__members__:
                value = enum_class[value]
            else:
                value = enum_class(value)
        key[column] = value
    rules = [
        rule
        for rule in session.exec(select(ArchiveRule)).all()
        if cls.matches_rule(rule, key)
    ]
    return sorted(
        rules,
        key=lambda rule: (rule.table_name, str(rule.match), str(rule.successor)),
    )

    @classmethod
def hash(
    cls,
    session: Session,
    ignored: List[str] = ["observed_at"],
    progress: Optional[Progress] = None,
) -> dict:
    """Hash the content of the rows.

    Args:
        session: Database connection to use for object lookups.
        ignored: List of column names to exclude from hashing.
        progress: Optional progress bar to track the status of the hashing.

    Returns:
        Dictionary of the row hashes keyed by the JSON dump of primary keys.
    """
    pks = sorted(cls.get_columns()["primary_keys"])
    rows = session.exec(statement=select(cls))
    row_count = session.exec(select(func.count()).select_from(cls)).one()
    if progress:
        table_task_id = progress.add_task(
            cls.get_table_name(),
            total=row_count,
        )
    # no use of a generator as will need to serialize to JSON anyway
    hashes = {}
    for i, row in enumerate(rows):
        # NOTE Pydantic is warning when read Accelerator/Storage as dict
        # https://github.com/tiangolo/sqlmodel/issues/63#issuecomment-1081555082
        rowdict = row.model_dump(warnings=False)
        keys_id = cls.pk_key(rowdict)
        for dropkey in [*ignored, *pks]:
            rowdict.pop(dropkey, None)
        rowhash = sha1(dumps(rowdict, sort_keys=True).encode()).hexdigest()
        hashes[keys_id] = rowhash
        if progress:
            # updating the progress bar is expensive, so limit with manu iterations
            if row_count > 1e3:
                if (i + 1) % 1000 == 0:
                    progress.update(table_task_id, advance=1000)
                if i == row_count - 1:
                    progress.update(table_task_id, advance=row_count % 1000)
            else:
                progress.update(table_task_id, advance=1)

    return hashes

    class ResourceIdComparator(Comparator):
    """Compile server_id/database_id to CASE + typed filter."""

    def __init__(self, cls, resource_type, label):
        self.cls = cls
        self.resource_type = resource_type
        self.label = label

    def __clause_element__(self):
        return case(
            (self.cls.resource_type == self.resource_type, self.cls.resource_id),
            else_=None,
        ).label(self.label)

    def operate(self, op, *other, **kwargs):
        return and_(
            self.cls.resource_type == self.resource_type,
            op(self.cls.resource_id, *other, **kwargs),
        )

    # Scores refer to the resource of each ResourceType without a foreign key:
    # the value of the type is the parent table, and `<value>_id` its id column.

    @classmethod

        """Parent tables of the scores, including the resources they refer to."""
        return super().get_parent_tables() | {rt.value for rt in ResourceType}

    @classmethod
    def get_key_filters(cls, rule_table_name: str, keys: dict) -> Optional[dict]:
        """Map a rule on a resource onto `resource_type` and `resource_id`."""
        filters = super().get_key_filters(rule_table_name, keys)
        # membership via `in` raises TypeError for plain strings on Python 3.11
        resource_type = {member.value: member for member in ResourceType}.get(
            rule_table_name
        )
        if filters is None or resource_type is None:
            return filters
        filters["resource_type"] = resource_type
        id_column = f"{rule_table_name}_id"
        if id_column in filters:
            filters["resource_id"] = filters.pop(id_column)
        return filters
